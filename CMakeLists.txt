cmake_minimum_required(VERSION 3.20)
project(poison3d C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_EXTENSIONS OFF)

option(BUILD_WIN_SDL "Build Windows SDL2 app" ON)
option(BUILD_N64 "Generate N64 sources + build via libdragon (Docker)" OFF)
option(BUILD_DC "Generate Dreamcast sources + build via KOS (WSL)" OFF)

# Compiled into every platform's build
set(ENGINE_COMMON_SOURCES               
  src/engine/core/camera.c
  src/engine/core/engine.c
  src/engine/renderers/renderer_common.c
  src/engine/core/poison_filesystem.c
  src/engine/core/system_manager.c
)

# Shared app entry point, every platform
set(APP_COMMON_SOURCES                  
  src/apps/sandbox/main.c
  src/apps/sandbox/game_input_profile.c
)

# Win32 specific sources
set(ENGINE_WIN_SOURCES
  src/engine/input/mouse.c
  src/engine/images/image_format_tga.c
  src/engine/renderers/renderer_gl.c
  src/engine/systems/system_input_win_sdl.c
)
set(APP_WIN_SOURCES
  src/platforms/platform_win_sdl.c
)

# N64 specific sources
set(ENGINE_N64_SOURCES
  src/engine/renderers/renderer_n64.c
  src/engine/systems/system_input_n64.c
)
set(APP_N64_SOURCES
  src/platforms/platform_n64.c
)

# Dreamcast specific sources
set(ENGINE_DC_SOURCES
  src/engine/renderers/renderer_kgl.c
  src/engine/systems/system_input_kos.c
)

  src/platforms/platform_kos.c
)

# Produces generated makefiles for N64 and Dreamcast.
# We do this to keep all source files to build in this CMakeLists file
# while still building with N64Makefile and KosMakefile.
function(export_source_fragment out_file var_name sources strip_src_prefix)
  set(_rel_list "")

  foreach(_src ${sources})
    if(strip_src_prefix)
      string(REGEX REPLACE "^src/" "" _src "${_src}")
    endif()
    list(APPEND _rel_list "${_src}")
  endforeach()

  list(JOIN _rel_list " \\\n\t" _joined)

  file(GENERATE OUTPUT "${out_file}" CONTENT
"# AUTO-GENERATED by CMakeLists.txt at configure time -- do not edit by hand.
# Edit the ENGINE_*_SOURCES / APP_*_SOURCES lists in CMakeLists.txt instead,
# then re-run: cmake --preset <name>
${var_name} = \\
\t${_joined}
"
  )
endfunction()

if(BUILD_WIN_SDL)
  add_library(engine_core
    ${ENGINE_COMMON_SOURCES}
    ${ENGINE_WIN_SOURCES}
  )

  target_include_directories(engine_core PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src/engine/core
    ${CMAKE_CURRENT_SOURCE_DIR}/src/engine/renderers
  )

  add_executable(sandbox
    ${APP_COMMON_SOURCES}
    ${APP_WIN_SOURCES}
  )

  target_include_directories(sandbox PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/platforms
  )

  target_link_libraries(sandbox PRIVATE engine_core)

  # Use SDL2 via vcpkg manifest if available, else try system SDL2
  find_package(SDL2 QUIET)
  if(SDL2_FOUND AND TARGET SDL2::SDL2)
    target_link_libraries(sandbox PRIVATE SDL2::SDL2)
    target_link_libraries(engine_core PRIVATE SDL2::SDL2)
  else()
    target_compile_definitions(sandbox PRIVATE SDL_MAIN_HANDLED)
    target_link_libraries(sandbox PRIVATE SDL2)
    target_link_libraries(engine_core PRIVATE SDL2)
  endif()

  find_package(OpenGL REQUIRED)

  # Link to both OpenGL and GLU
  target_link_libraries(sandbox PRIVATE OpenGL::GLU OpenGL::GL)

  # Link against the Windows OpenGL loader (provides gl 1.1)
  if(WIN32)
    target_link_libraries(sandbox PRIVATE opengl32)
  endif()

  if(MSVC)
    target_compile_options(sandbox PRIVATE /W4 /permissive-)
    target_compile_options(engine_core PRIVATE /W4 /permissive-)
  else()
    target_compile_options(sandbox PRIVATE -Wall -Wextra -Werror=return-type)
    target_compile_options(engine_core PRIVATE -Wall -Wextra -Werror=return-type)
  endif()
endif()

if(BUILD_N64)
  set(_n64_sources
    ${ENGINE_COMMON_SOURCES}
    ${APP_COMMON_SOURCES}
    ${ENGINE_N64_SOURCES}
    ${APP_N64_SOURCES}
  )
  export_source_fragment(
    "${CMAKE_SOURCE_DIR}/generated/n64_sources.mk"
    GEN_N64_SOURCES
    "${_n64_sources}"
    TRUE
  )

  add_custom_target(n64 ALL
    COMMAND cmd.exe /c "libdragon make -f N64Makefile"
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "Building N64 ROM via libdragon (Docker), sources from generated/n64_sources.mk"
    VERBATIM
  )
endif()

if(BUILD_DC)
  set(_dc_sources
    ${ENGINE_COMMON_SOURCES}
    ${APP_COMMON_SOURCES}
    ${ENGINE_DC_SOURCES}
    ${APP_DC_SOURCES}
  )
  export_source_fragment(
    "${CMAKE_SOURCE_DIR}/generated/dreamcast_sources.mk"
    GEN_DC_SOURCES
    "${_dc_sources}"
    FALSE
  )

  add_custom_target(dreamcast ALL
    COMMAND wsl.exe bash -lc ". /opt/toolchains/dc/kos/environ.sh && make clean -f KosMakefile && make -f KosMakefile"
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "Building Dreamcast ELF via KOS (WSL), sources from generated/dreamcast_sources.mk"
    VERBATIM
  )
endif()
