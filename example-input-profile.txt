# A profile is defined, such as "UI" or "Gameplay".
# The profile may be switched 
#
# profile_name - how actions refer to this profile
# platforms - comma separated platform selection, 
# 	      may be dc, n64, win, lin, osx
#
[profile(profile_name)]
platforms=win,n64

# type - may be 1f, 2f, or performed
# profile_name - the profile this action will bind to
# action_name - how game logic may query for if this action was performed
# example:
#
#	[profile(GameplayProfile)]
#	platforms=win,n64
#
#	[action(2f, GameplayProfile, Walk)]
#
[action(type, profile_name, action_name)]

[action_binding(action_name, axis_up, axis_down, axis_left, axis_right)]
